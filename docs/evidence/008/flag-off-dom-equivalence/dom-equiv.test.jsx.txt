import { describe, expect, it, vi } from "vitest";
import { fireEvent, render, screen, waitFor } from "@testing-library/react";
import fs from "node:fs";
import App from "../App";
const gs = vi.fn();
vi.mock("../billingClient", async (o) => ({ ...(await o()), readCustomerBillingState: vi.fn().mockResolvedValue({ status: "not_subscribed", subscription: null }) }));
vi.mock("../session", () => ({ getCustomerSession: (...a) => gs(...a), refreshCustomerSession: vi.fn() }));
vi.mock("../authClient", () => ({ onAuthStateChange: vi.fn(() => () => {}), signOut: vi.fn(), signInWithPassword: vi.fn(), signUpWithPassword: vi.fn() }));
const ok = (b) => ({ ok: true, status: 200, headers: new Headers(), json: async () => b });
const CID = "22222222-2222-4222-8222-222222222222";
const REC = { recommendation_id: "r", tenant_id: "t", project_id: "p", brand_id: "b", campaign_name: "Spring Campaign", summary: "S", explanation: "E", create_campaign_payload: { tenant_id: "t", project_id: "p", brand_id: "b", name: "Spring Campaign", goal: "g", display_timezone: "Europe/London" }, suggested_items: [{ name: "Offer", format: "text", platform: "instagram", placement: "feed", destination_label: "Instagram", reason: "r" }] };
const camp = (v, n) => ({ campaign_id: CID, name: "Spring Campaign", version: v, items: REC.suggested_items.slice(0, n).map((i, k) => ({ content_item_id: "c" + k, name: i.name, format: i.format, variants: [{ variant_id: "v" + k, destination_label: i.destination_label, workflow: "draft" }] })) });
describe("flag-off DOM snapshots", () => {
  it("captures", async () => {
    const out = {};
    for (const [label, sess] of [["signed-out", { status: "signed-out" }], ["unconfigured", { status: "unconfigured" }], ["scope-missing", { status: "scope-missing", accessToken: "t" }], ["ready", { status: "ready", accessToken: "t", tenantId: "t", projectId: "p", brandId: "b" }]]) {
      gs.mockResolvedValue(sess); let n = 0;
      global.fetch = vi.fn(async (url, o = {}) => { const p = new URL(url).pathname; const m = (o.method || "GET");
        if (p.endsWith("/brand-brain")) return ok({ status: "ready", brand_brain: { name: "X", identity: { description: "d" }, metadata: { version: 1, status: "draft" } } });
        if (p.endsWith("/campaign-recommendations")) return ok({ recommendation: REC });
        if (p.endsWith("/campaigns") && m === "POST") return ok({ result: { campaign_id: CID, campaign_version: 1 }, campaign: camp(1, 0) });
        if (p.endsWith("/content-items")) { n++; return ok({ campaign: camp(n + 1, n) }); }
        return { ok: false, status: 404, headers: new Headers(), json: async () => ({}) }; });
      const { container, unmount } = render(<App />);
      await waitFor(() => expect(container.querySelector(".topbar")).toBeTruthy()); await new Promise((r) => setTimeout(r, 30));
      out[label + ":initial"] = container.innerHTML;
      if (label === "ready") {
        fireEvent.click(await screen.findByRole("button", { name: /load brand brain/i }).catch(() => screen.getByRole("button", { name: /brand brain/i })));
        await new Promise((r) => setTimeout(r, 30)); out[label + ":brain"] = container.innerHTML;
        fireEvent.change(screen.getByLabelText(/your goal/i), { target: { value: "Promote the spring offer" } });
        fireEvent.click(screen.getByRole("button", { name: /get my recommendation/i })); await screen.findByText("Spring Campaign"); out[label + ":recommendation"] = container.innerHTML;
        fireEvent.click(screen.getByRole("button", { name: /^create campaign$/i })); await screen.findByText(/saved to your workspace/i); out[label + ":campaign"] = container.innerHTML;
      }
      unmount();
    }
    fs.writeFileSync(process.env.DOM_OUT, JSON.stringify(out, null, 1));
    expect(Object.keys(out).length).toBeGreaterThan(5);
  });
});
